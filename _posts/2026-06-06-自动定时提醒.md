用shell写一个在系统后台自动运行的定时提醒脚本，通过pushover转发消息到手机。

第一步：准备一个任务设定配置文件tasks.conf。可以设定提醒时间形式，消息内容，优先级，链接(苹果系统的快捷指令)，收到提醒的系统声音，提前提醒天数等。代码如下：
~~~
# 类型   参数1     参数2    MESSAGE                         SHORTCUT_URL     PRIORITY  SOUND  ADVANCE
# ----------------------------------------------------------------------------------------

# A. 每个工作日的提醒（排除联邦 + BC 省假日）其中提醒级别是2，会重复提醒直到点击确认（可以设定持续多久，以6.5小时为例）  
workday 06:28 - 注意情绪稳定 shortcuts://run-shortcut?name=定时朗读 priority=2 
workday 07:45, 08:45, 09:45, 10:45, 11:45, 12:45 - 已操作一小时_休息喝水 shortcuts://run-shortcut?name=休息一下 priority=0 sound=magic

# B. 每天 
# 睡觉提醒（不排除节假日）
daily 22:00 - 该睡觉了_准备休息 shortcuts://run-shortcut?name=准备睡觉 priority=0
# 喝水休息提醒（不排除节假日）
daily 14:00, 15:00, 16:00, 17:00, 18:00, 19:00, 20:00, 21:00 - 休息_站起来转一转拉伸一下身体喝口水 shortcuts://run-shortcut?name=休息一下 priority=0

# C. 每周提醒
# 周四定期活动（不排除节假日）
4 17:00 - 参加每周活动 none priority=0

# D. 每月提醒
# 信用卡还款提醒
monthly 15 10:00 注意查看几张信用卡还款情况 none priority=2

# E. 每年提醒
# 生日提醒
yearly 03-10 19:00 妈妈生日3/10到了 none priority=0 advance=7d
# 纪念等特殊日提醒
yearly 06-04 13:00 生日6/4要到了 none priority=2 advance=1d
# 年续费提醒
yearly 11-03 19:00 Tradingview马上要续费了 none priority=2
# 报税提醒
yearly 04-21 19:00 报税要不能再拖了 none priority=2
~~~

有关节假日可以用脚本从网上自动抓去。首先要安装一个jq的库：`sudo apt install jq -y`，然后编写脚本代码如下（update_holidays.sh）：
~~~
#!/bin/bash

HOLIDAY_JSON="/home/ubuntu/automation/holidays.json"
TMP1="/home/ubuntu/automation/holidays_year1.json"
TMP2="/home/ubuntu/automation/holidays_year2.json"

YEAR=$(date +%Y)
NEXT_YEAR=$((YEAR + 1))

# 下载今年和明年的节假日
curl -s "https://canada-holidays.ca/api/v1/holidays?year=$YEAR" -o "$TMP1"
curl -s "https://canada-holidays.ca/api/v1/holidays?year=$NEXT_YEAR" -o "$TMP2"

# 使用 jq 提取 .holidays 并合并两个数组
jq -s '
  # 取出每个文件中的 .holidays 数组
  map(.holidays) 
  # 合并成一个大数组
  | add
  # 只保留我们需要的字段
  | map({
      date: .date,
      name: .nameEn,
      federal: (.federal == 1),
      provinces: [.provinces[].id]
    })
' "$TMP1" "$TMP2" > "$HOLIDAY_JSON"

echo "Updated: $HOLIDAY_JSON"
~~~

每天自动更新（cron）节假日最新信息：
`sudo chmod +x /home/ubuntu/automation/update_holidays.sh`
`sudo crontab -e`
添加一行（每天凌晨 3 点更新，可以不用这么频繁更新）：
`0 3 * * * /home/ubuntu/automation/update_holidays.sh`

第二步：提醒脚本reminder.sh，代码如下：
~~~
#!/bin/bash

BASE_DIR="/home/ubuntu/automation" #这里是代码存放的位置
CONFIG_FILE="$BASE_DIR/tasks.conf"
HOLIDAY_FILE="$BASE_DIR/holidays.json"
LOG_FILE="$BASE_DIR/reminder.log"
LAST_CLEAN_FILE="$BASE_DIR/.last_log_clean"

# ====== Pushover 配置（改成你自己的）======
PUSHOVER_USER_KEY="这里填写pushover的user_key" #这里需要修改更新
PUSHOVER_API_TOKEN="这里填写pushover的API_TOKEN" #这里需要修改更新

# >>> NEW: 默认 retry/expire（仅在 priority=2 时使用）
PUSHOVER_RETRY=1800
PUSHOVER_EXPIRE=23400

# >>> NEW: send_pushover 增强，支持 priority + sound
send_pushover() {
    local MESSAGE="$1"
    local URL="$2"
    local PRIORITY="$3"
    local SOUND="$4"

    if [[ -n "$PUSHOVER_USER_KEY" && -n "$PUSHOVER_API_TOKEN" ]]; then

        # >>> NEW: 紧急模式（priority=2）需要 retry/expire，sound 被忽略
        if [[ "$PRIORITY" == "2" ]]; then
            if [[ "$URL" != "none" && -n "$URL" ]]; then
                curl -s \
                    -F "token=$PUSHOVER_API_TOKEN" \
                    -F "user=$PUSHOVER_USER_KEY" \
                    -F "message=$MESSAGE" \
                    -F "priority=$PRIORITY" \
                    -F "retry=$PUSHOVER_RETRY" \
                    -F "expire=$PUSHOVER_EXPIRE" \
                    -F "callback=https://mkzg.com" \
                    -F "url=$URL" \
                    # -F "url_title=打开链接" \
                    https://api.pushover.net/1/messages.json > /dev/null
            else
                curl -s \
                    -F "token=$PUSHOVER_API_TOKEN" \
                    -F "user=$PUSHOVER_USER_KEY" \
                    -F "message=$MESSAGE" \
                    -F "priority=$PRIORITY" \
                    -F "retry=$PUSHOVER_RETRY" \
                    -F "expire=$PUSHOVER_EXPIRE" \
                    https://api.pushover.net/1/messages.json > /dev/null
            fi
            return
        fi

        # >>> NEW: 普通模式（priority -2/-1/0/1）支持 sound
        if [[ "$URL" != "none" && -n "$URL" ]]; then
            curl -s \
                -F "token=$PUSHOVER_API_TOKEN" \
                -F "user=$PUSHOVER_USER_KEY" \
                -F "message=$MESSAGE" \
                -F "priority=$PRIORITY" \
                -F "sound=$SOUND" \
                -F "url=$URL" \
                # -F "url_title=打开链接" \
                https://api.pushover.net/1/messages.json > /dev/null
        else
            curl -s \
                -F "token=$PUSHOVER_API_TOKEN" \
                -F "user=$PUSHOVER_USER_KEY" \
                -F "message=$MESSAGE" \
                -F "priority=$PRIORITY" \
                -F "sound=$SOUND" \
                https://api.pushover.net/1/messages.json > /dev/null
        fi
    fi
}

log() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') $1" >> "$LOG_FILE"
}

clean_log_daily() {
    local TODAY
    TODAY=$(date +%Y-%m-%d)

    if [[ ! -f "$LAST_CLEAN_FILE" ]]; then
        echo "$TODAY" > "$LAST_CLEAN_FILE"
        return
    fi

    local LAST_CLEAN
    LAST_CLEAN=$(cat "$LAST_CLEAN_FILE")

    if [[ "$LAST_CLEAN" != "$TODAY" ]]; then
        : > "$LOG_FILE"
        echo "$TODAY" > "$LAST_CLEAN_FILE"
        echo "$(date '+%Y-%m-%d %H:%M:%S') Log cleaned for new day" >> "$LOG_FILE"
    fi
}

is_holiday_today() {
    [[ ! -f "$HOLIDAY_FILE" ]] && return 1
    local TODAY
    TODAY=$(date +%Y-%m-%d)
    jq -e ".[] | select(.date == \"$TODAY\")" "$HOLIDAY_FILE" > /dev/null 2>&1
}

is_workday_today() {
    local DOW
    DOW=$(date +%u)
    if (( DOW >= 1 && DOW <= 5 )); then
        if is_holiday_today; then
            return 1
        else
            return 0
        fi
    else
        return 1
    fi
}

# >>> NEW: notify 增强，支持 sound
notify() {
    local MESSAGE="$1"
    local URL="$2"
    local PRIORITY="$3"
    local SOUND="$4"

    if [[ "$URL" != "none" && -n "$URL" ]]; then
        /usr/bin/osascript <<EOF
tell application "Safari"
    open location "$URL"
end tell
display notification "$MESSAGE" with title "Reminder"
EOF
    else
        /usr/bin/osascript -e "display notification \"$MESSAGE\" with title \"Reminder\""
    fi

    send_pushover "$MESSAGE" "$URL" "$PRIORITY" "$SOUND"
}

while true; do
    clean_log_daily

    NOW=$(date +%H:%M)
    TODAY_WEEK=$(date +%u)
    TODAY_DAY=$(date +%d)
    TODAY_MD=$(date +%m-%d)
    TODAY_FULL=$(date +%Y-%m-%d)   # >>> NEW

    while IFS= read -r LINE; do
        LINE="${LINE#"${LINE%%[![:space:]]*}"}"
        LINE="${LINE%"${LINE##*[![:space:]]}"}"

        [[ -z "$LINE" ]] && continue
        [[ "$LINE" =~ ^# ]] && continue

        # >>> NEW: 解析 priority（可选）
        PRIORITY=0
        if [[ "$LINE" =~ priority=([0-2]|-1|-2) ]]; then
            PRIORITY="${BASH_REMATCH[1]}"
        fi

        # >>> NEW: 解析 sound（可选）
        SOUND="default"
        if [[ "$LINE" =~ sound=([a-zA-Z0-9_-]+) ]]; then
            SOUND="${BASH_REMATCH[1]}"
        fi

        # >>> NEW: 解析 advance=Xd（可选）
        ADVANCE_DAYS=0
        if [[ "$LINE" =~ advance=([0-9]+)d ]]; then
            ADVANCE_DAYS="${BASH_REMATCH[1]}"
        fi

        # ---------- workday ----------
        if [[ "$LINE" =~ ^workday[[:space:]]+ ]]; then
            REST="${LINE#workday }"
            TIMES_PART="${REST%% - *}"
            AFTER_DASH="${REST#* - }"
            URL="${AFTER_DASH##* }"
            MESSAGE="${AFTER_DASH% *}"

            if ! is_workday_today; then continue; fi

            IFS=',' read -ra TIMES <<< "$TIMES_PART"
            for T in "${TIMES[@]}"; do
                T=$(echo "$T" | xargs)
                if [[ "$NOW" == "$T" ]]; then
                    log "workday: $MESSAGE ($URL)"
                    notify "$MESSAGE" "$URL" "$PRIORITY" "$SOUND"
                    sleep 60
                fi
            done
            continue
        fi

        # ---------- daily ----------
        if [[ "$LINE" =~ ^daily[[:space:]]+ ]]; then
            REST="${LINE#daily }"
            TIMES_PART="${REST%% - *}"
            AFTER_DASH="${REST#* - }"
            URL="${AFTER_DASH##* }"
            MESSAGE="${AFTER_DASH% *}"

            IFS=',' read -ra TIMES <<< "$TIMES_PART"
            for T in "${TIMES[@]}"; do
                T=$(echo "$T" | xargs)
                if [[ "$NOW" == "$T" ]]; then
                    log "daily: $MESSAGE ($URL)"
                    notify "$MESSAGE" "$URL" "$PRIORITY" "$SOUND"
                    sleep 60
                fi
            done
            continue
        fi

        # ---------- weekly ----------
        if [[ "$LINE" =~ ^[1-7][[:space:]]+ ]]; then
            WEEK="${LINE%% *}"
            REST="${LINE#* }"
            TIMES_PART="${REST%% - *}"
            AFTER_DASH="${REST#* - }"
            URL="${AFTER_DASH##* }"
            MESSAGE="${AFTER_DASH% *}"

            if [[ "$WEEK" != "$TODAY_WEEK" ]]; then continue; fi

            IFS=',' read -ra TIMES <<< "$TIMES_PART"
            for T in "${TIMES[@]}"; do
                T=$(echo "$T" | xargs)
                if [[ "$NOW" == "$T" ]]; then
                    log "weekly($WEEK): $MESSAGE ($URL)"
                    notify "$MESSAGE" "$URL" "$PRIORITY" "$SOUND"
                    sleep 60
                fi
            done
            continue
        fi

        # ---------- monthly ----------
        if [[ "$LINE" =~ ^monthly[[:space:]]+ ]]; then
            REST="${LINE#monthly }"
            DAY="${REST%% *}"
            AFTER_DAY="${REST#* }"
            TIMES_PART="${AFTER_DAY%% *}"
            AFTER_TIME="${AFTER_DAY#* }"
            URL="${AFTER_TIME##* }"
            MESSAGE="${AFTER_TIME% *}"

            if [[ "$DAY" != "$TODAY_DAY" ]]; then continue; fi

            IFS=',' read -ra TIMES <<< "$TIMES_PART"
            for T in "${TIMES[@]}"; do
                T=$(echo "$T" | xargs)
                if [[ "$NOW" == "$T" ]]; then
                    log "monthly($DAY): $MESSAGE ($URL)"
                    notify "$MESSAGE" "$URL" "$PRIORITY" "$SOUND"
                    sleep 60
                fi
            done
            continue
        fi

        # ---------- yearly ----------
        if [[ "$LINE" =~ ^yearly[[:space:]]+ ]]; then
            REST="${LINE#yearly }"
            MD="${REST%% *}"
            AFTER_MD="${REST#* }"
            TIMES_PART="${AFTER_MD%% *}"
            AFTER_TIME="${AFTER_MD#* }"
            URL="${AFTER_TIME##* }"
            MESSAGE="${AFTER_TIME% *}"

            # >>> NEW: yearly + advance=Xd 支持
            if (( ADVANCE_DAYS > 0 )); then
                YEAR=$(date +%Y)
                EVENT_DATE_THIS_YEAR="$YEAR-$MD"
                TODAY_SEC=$(date -d "$TODAY_FULL" +%s)
                EVENT_SEC_THIS_YEAR=$(date -d "$EVENT_DATE_THIS_YEAR" +%s)

                if (( EVENT_SEC_THIS_YEAR < TODAY_SEC )); then
                    EVENT_YEAR=$((YEAR + 1))
                else
                    EVENT_YEAR=$YEAR
                fi

                EVENT_DATE="$EVENT_YEAR-$MD"
                REMIND_DATE=$(date -d "$EVENT_DATE - $ADVANCE_DAYS day" +%Y-%m-%d)
                REMIND_MD=$(date -d "$REMIND_DATE" +%m-%d)

                if [[ "$REMIND_MD" != "$TODAY_MD" ]]; then
                    continue
                fi
            else
                # 原有逻辑：当天提醒
                if [[ "$MD" != "$TODAY_MD" ]]; then
                    continue
                fi
            fi

            IFS=',' read -ra TIMES <<< "$TIMES_PART"
            for T in "${TIMES[@]}"; do
                T=$(echo "$T" | xargs)
                if [[ "$NOW" == "$T" ]]; then
                    log "yearly($MD): $MESSAGE ($URL)"
                    notify "$MESSAGE" "$URL" "$PRIORITY" "$SOUND"
                    sleep 60
                fi
            done
            continue
        fi

    done < "$CONFIG_FILE"

    sleep 10
done
~~~

第三步：设置开机自动启动脚本的systemd服务
创建启动服务文件：/etc/systemd/system/reminder.service
~~~
[Unit]
Description=Personal Reminder Service
After=network.target

[Service]
ExecStart=/home/ubuntu/automation/reminder.sh
Restart=always
RestartSec=5
StandardOutput=append:/var/log/reminder.log
StandardError=append:/var/log/reminder.log

[Install]
WantedBy=multi-user.target
~~~

第四步：启用
sudo chmod +x /home/ubuntu/automation/reminder.sh
sudo systemctl daemon-reload
sudo systemctl enable reminder
sudo systemctl start reminder
查看状态（可选）：
systemctl status reminder
tail -f /var/log/reminder.log

以后只需要做的事
- 想到加新的定期提醒 → 在 /home/ubuntu/automation/tasks.conf 里加新的一行提醒任务配置，加完之后执行一下 sudo systemctl restart reminder.service 重新启动（重新读取新的配置文件）。
- 节假日信息 → update_holidays.sh + cron 自动维护 holidays.json
